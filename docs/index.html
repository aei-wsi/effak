Quick Start

# Clone the repository
git clone https://github.com/aei-wsi/effak.git
cd effak/backend

# Install dependencies
npm install

# Configure environment
cp .env.example .env
# Edit .env with your settings

# Create database
psql -U postgres -c "CREATE DATABASE effak;"

# Run migrations
npm run migrate

# Start server
npm run dev

# Server running at http://localhost:5000
